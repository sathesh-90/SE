import java.util.*;

class Solution {

    // Function to find all combinations
    public List<List<Integer>> combinationSum(int[] candidates, int target) {

        List<List<Integer>> ans = new ArrayList<>();
        List<Integer> path = new ArrayList<>();
        int n = candidates.length;

        dfs(0, target, candidates, n, path, ans);

        return ans;
    }

    // Backtracking function
    private void dfs(int index, int remaining, int[] candidates, int n,
                     List<Integer> path, List<List<Integer>> ans) {

        // If target becomes 0, store the current combination
        if (remaining == 0) {
            ans.add(new ArrayList<>(path));
            return;
        }

        // Stop if target becomes negative or all candidates are processed
        if (remaining < 0 || index == n) {
            return;
        }

        // -------------------------
        // Choice 1: Pick current candidate
        // -------------------------
        path.add(candidates[index]);

        // Stay at the same index because
        // the same number can be used multiple times
        dfs(index, remaining - candidates[index], candidates, n, path, ans);

        // Backtrack
        path.remove(path.size() - 1);

        // -------------------------
        // Choice 2: Skip current candidate
        // -------------------------
        dfs(index + 1, remaining, candidates, n, path, ans);
    }
}

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        // Number of candidates
        int n = sc.nextInt();

        // Candidates array
        int[] candidates = new int[n];

        for (int i = 0; i < n; i++) {
            candidates[i] = sc.nextInt();
        }

        // Target value
        int target = sc.nextInt();

        Solution obj = new Solution();

        List<List<Integer>> result = obj.combinationSum(candidates, target);

        System.out.println(result);

        sc.close();
    }
}
