import java.util.*;

/*
Algorithm:
1. Sort meetings by start time.
2. Maintain a min-heap storing end times of meetings currently occupying rooms.
3. For every meeting:
   - If its start time is strictly greater than the smallest end time,
     that room becomes free, so remove it from heap.
   - Add current meeting's end time into heap.
4. The maximum size of heap during processing is the minimum rooms required.

Time Complexity:
Sorting : O(N log N)
Heap Operations : O(N log N)
Overall : O(N log N)

Space Complexity:
O(N)
*/

public class Main {

    // Function to find minimum meeting rooms
    public static int minimumRooms(int[][] intervals) {

        // Sort meetings by starting time
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));

        // Min Heap stores ending times
        PriorityQueue<Integer> minHeap = new PriorityQueue<>();

        int answer = 0;

        // Process every meeting
        for (int[] meeting : intervals) {

            int start = meeting[0];
            int end = meeting[1];

            // Room becomes free only if current meeting starts AFTER previous ends
            if (!minHeap.isEmpty() && start > minHeap.peek()) {
                minHeap.poll();
            }

            // Allocate room
            minHeap.offer(end);

            // Track maximum rooms needed
            answer = Math.max(answer, minHeap.size());
        }

        return answer;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        int[][] intervals = new int[n][2];

        for (int i = 0; i < n; i++) {
            intervals[i][0] = sc.nextInt();
            intervals[i][1] = sc.nextInt();
        }

        System.out.println(minimumRooms(intervals));

        sc.close();
    }
}
